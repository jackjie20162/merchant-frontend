func (l *MerchantLogic) FreightSave(req *FreightSaveReq) (*wssclient.BaseIDUint32Resp, error) {
	if req.Freight == nil {
		return nil, errors.New("运费模板信息不能为空")
	}
	shopID := l.shopID()
	req.Freight.ShopId = ptrUint32(shopID)
	var freightID uint32
	if req.Freight.Id == nil || *req.Freight.Id == 0 {
		resp, err := l.svcCtx.WssRpc.CreateShopFreight(l.ctx, req.Freight)
		if err != nil {
			return nil, err
		}
		freightID = resp.GetId()
	} else {
		freightID = *req.Freight.Id
		old, err := l.svcCtx.WssRpc.GetShopFreightById(l.ctx, &wssclient.IDUint32Req{Id: freightID})
		if err != nil {
			return nil, err
		}
		if old.GetShopId() != shopID {
			return nil, errNotOwned
		}
		if _, err = l.svcCtx.WssRpc.UpdateShopFreight(l.ctx, req.Freight); err != nil {
			return nil, err
		}
	}
	if err := l.syncFreightData(freightID, req.Data); err != nil {
		return nil, err
	}
	return &wssclient.BaseIDUint32Resp{Id: freightID}, nil
}
