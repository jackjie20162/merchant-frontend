func (l *MerchantLogic) WithdrawApply(req *wssclient.WithdrawInfo) (*wssclient.BaseIDUint32Resp, error) {
	if req.Money == nil || *req.Money <= 0 {
		return nil, errors.New("提现金额必须大于0")
	}
	summary, err := l.svcCtx.WssRpc.GetShopFinanceSummary(l.ctx, &wssclient.ShopFinanceSummaryReq{ShopId: l.shopID()})
	if err != nil {
		return nil, err
	}
	if *req.Money > summary.GetBalance() {
		return nil, errors.New("提现金额超过可提现余额")
	}
	req.Id = nil
	req.UserId = nil
	req.Username = nil
	req.Status = nil
	req.ShopId = ptrUint32(l.shopID())
	return l.svcCtx.WssRpc.CreateWithdraw(l.ctx, req)
}
