// staticFileHandler 上传附件静态文件服务（GET /static/:ymd/:file）
func staticFileHandler(svcCtx *svc.ServiceContext) http.HandlerFunc {
	uploadDir := svcCtx.Config.UploadDir
	if uploadDir == "" {
		uploadDir = "./uploads"
	}
	absDir, err := filepath.Abs(uploadDir)
	if err != nil {
		absDir = uploadDir
	}
	return func(w http.ResponseWriter, r *http.Request) {
		var req struct {
			Ymd  string `path:"ymd"`
			File string `path:"file"`
		}
		if err := httpx.Parse(r, &req, true); err != nil {
			httpx.ErrorCtx(r.Context(), w, err)
			return
		}
		// 防路径穿越：段内不允许 ..
		if strings.Contains(req.Ymd, "..") || strings.Contains(req.File, "..") {
			http.NotFound(w, r)
			return
		}
		http.ServeFile(w, r, filepath.Join(absDir, req.Ymd, req.File))
	}
}
