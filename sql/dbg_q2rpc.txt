// parseJsonBodyLimit 涓?parseJsonBody 鐩稿悓锛屼絾鍏佽鑷畾涔?body 涓婇檺
// 锛堝晢鍝佷繚瀛?瑁呬慨椤甸潰 JSON 鍙兘瓒呰繃榛樿 1MB锛?func parseJsonBodyLimit(r *http.Request, v any, max int64) error {
	body, err := io.ReadAll(io.LimitReader(r.Body, max))
	if err != nil {
		return err
	}
	if len(body) == 0 {
		return nil
	}
	return json.Unmarshal(body, v)
}

	var req struct {
		Id uint32 `path:"id"`
	}
	if err := httpx.Parse(r, &req, true); err != nil {
		return 0, err
	}
	return req.Id, nil
}
