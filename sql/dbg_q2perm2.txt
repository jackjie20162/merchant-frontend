// allPermCodes 返回全部权限码（主账号拥有全部权限）
func allPermCodes() []string {
	codes := make([]string, 0, len(permCodes))
	for _, p := range permCodes {
		codes = append(codes, p.code)
	}
	return codes
}

// permCodeListResp 返回权限码注册表响应
func permCodeListResp() *wss.PermCodeListResp {
	resp := &wss.PermCodeListResp{}
	for _, p := range permCodes {
		resp.Data = append(resp.Data, &wss.PermCodeInfo{
			Code:   p.code,
			Name:   p.name,
			Module: p.module,
		})
	}
	return resp
}

// validPerms 校验权限码是否都在注册表中，返回非法的权限码
func invalidPerms(codes []string) []string {
	registry := make(map[string]struct{}, len(permCodes))
	for _, p := range permCodes {
		registry[p.code] = struct{}{}
	}
	var invalid []string
	for _, c := range codes {
		if _, ok := registry[c]; !ok {
			invalid = append(invalid, c)
		}
	}
	return invalid
}

// permsToString 权限码切片转逗号分隔字符串
func permsToString(codes []string) string {
	return strings.Join(codes, ",")
}

// permsFromString 逗号分隔字符串转权限码切片
func permsFromString(s string) []string {
	if s == "" {
		return nil
	}
	parts := strings.Split(s, ",")
	codes := make([]string, 0, len(parts))
	for _, p := range parts {
		if p = strings.TrimSpace(p); p != "" {
			codes = append(codes, p)
		}
	}
	return codes
}

// md5WithSalt returns FastAdmin style hash: md5(md5(plain) + salt)
func md5WithSalt(plain, salt string) string {
	first := md5.Sum([]byte(plain))
	second := md5.Sum([]byte(hex.EncodeToString(first[:]) + salt))
	return hex.EncodeToString(second[:])
}

// newSalt 生成随机密码盐
func newSalt() string {
	return uuid.NewString()[:8]
}
